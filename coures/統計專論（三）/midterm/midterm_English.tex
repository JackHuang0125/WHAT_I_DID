\begin{frame}{Why Is Estimation Important in SPC?}
    In SPC, the in-control state of a process is usually described by baseline parameters such as $\mu_0,\, \sigma_0$

    \vspace{0.8em}
    However, in practice these quantities are usually \textbf{unknown} and must be estimated from Phase I historical data

    \vspace{0.8em}
    \textbf{Key points:}
        \begin{itemize}
            \item If $\hat{\mu}_0$ is estimated incorrectly, the center line will shift
            \item If $\hat{\sigma}_0$ is estimated incorrectly, the control limits will become too wide or too narrow
            \item Incorrect limits may lead to: 
            \begin{itemize}
                \item Too many false alarms
                \item Delayed detection of abnormal changes
            \end{itemize}
        \end{itemize}

    Therefore, SPC is not only about drawing charts. The first question is:

    \uncover<2->{
    {\centering
    \Huge\textbf{How good is the estimator?}\par}
    }
\end{frame}

%================================================
\begin{frame}{Estimators and Sampling Distributions}
    Suppose $$X_1, \dots,X_n \overset{iid}{\sim} F_\theta$$ where $\theta$ is an unknown parameter

    \vspace{0.5em}
    An \textbf{estimator}is a function of the sample:
    $$\hat{\theta}=T(X_1, \dots,X_n)$$ 
    Since $X_1, \dots, X_n$ are random, $\hat{\theta}$ is also a random variable

    Therefore, we should not only ask “what value do we get?”, but also ask:
        \begin{itemize}
            \item What is $E(\hat{\theta})$?
            \item What is $\mathrm{Var}(\hat{\theta})$?
            \item Does $\hat{\theta}$ converge to the true value $\theta$ as $n \to \infty$?
        \end{itemize}
    % \vspace{0.8em}
    % \textbf{這三個問題對應到: }
    % $$
    % \text{Bias}, \quad \text{MSE}, \quad \text{Consistency}
    % $$
\end{frame}

%================================================
\begin{frame}{Bias: Does the Estimator Systematically Deviate from the Truth?}
    \textbf{Definition: }
    $$\mathrm{Bias}(\hat{\theta}, \theta)=E(\hat{\theta})-\theta$$ % 長期平均誤差
    If for all $\theta$
    $$E(\hat{\theta})=\theta$$ % 若長期平均誤差不偏, 則是unbiased
    then $\hat{\theta}$ is called an \textbf{unbiased estimator}

    \vspace{0.8em}
    \textbf{Interpretation: } Bias measures whether the estimator tends to overestimate or underestimate the true value on average
                    % Bias $>0$: 平均高估
                    % Bias $<0$: 平均低估
                    % Bias $=0$: 沒有系統性偏差
    
    \vspace{0.8em}
    \textbf{Example:}
    Suppose $E(X_i)=\mu$ and $\bar X=\frac1n\sum_{i=1}^n X_i.$ Then $E(\bar X)=\mu$
    
    \vspace{0.8em}
    Therefore, $\bar X$ is an unbiased estimator of $\mu$
\end{frame}

%================================================
\begin{frame}{MSE: Unbiasedness Is Not Enough}
    \vspace{-1em}
    The \textbf{Mean Squared Error} is defined as
    \vspace{-0.8em}
    $$\mathrm{MSE}(\hat{\theta}, \theta)=E\big[(\hat{\theta}-\theta)^2\big]$$
    % 單次估計誤差
    It can be \textbf{decomposed} into
    $$\boxed{\mathrm{MSE}(\hat{\theta}, \theta)=\mathrm{Var}(\hat{\theta})+\mathrm{Bias}^2(\hat{\theta}, \theta)}$$
    \textbf{Key message: }
    \begin{itemize}
        \item An estimator can be unbiased but still have very large variance
        \item A slightly biased estimator may have a much smaller MSE
    \end{itemize}
\end{frame}

%================================================
\begin{frame}{Example}
    Goal: Estimate $\mu$, assuming\,$X_1, \dots,X_n \sim (\mu, \sigma^2)$

    \vspace{0.5em}
    Consider two estimators: $\quad\hat{\mu}_1=\bar X, \quad\hat{\mu}_2=0.8\bar X$

    \vspace{0.5em}
    \textbf{First estimator:} $E(\hat{\mu}_1)=\mu, \quad\mathrm{Var}(\hat{\mu}_1)=\frac{\sigma^2}{n}, \quad\text{thus\quad}\mathrm{MSE}(\hat{\mu}_1)=\frac{\sigma^2}{n}$

    \vspace{0.5em}
    \textbf{Second estimator:}
    \vspace{-0.8em}
    \begin{align*}
    &E(\hat{\mu}_2)=0.8\mu, \quad\mathrm{Var}(\hat{\mu}_2)=0.8^2\mathrm{Var}(\bar X)=0.64\frac{\sigma^2}{n}\\
    &\mathrm{Bias}(\hat{\mu}_2, \mu)=0.8\mu-\mu=-0.2\mu, \\
    &\text{Therefore},\, \mathrm{MSE}(\hat{\mu}_2)=0.64\frac{\sigma^2}{n}+(0.2\mu)^2
    \end{align*}

    \textbf{Conclusion:}
    A biased estimator may still have a smaller MSE if its variance is sufficiently reduced
\end{frame}

\begin{frame}{Consistency: Does the Estimator Approach the True Value as the Sample Size Increases?}
    \textbf{Definition:}
    If a sequence of estimators $\hat{\theta}_n$ satisfies $$\hat{\theta}_n \xrightarrow{P} \theta$$
    then $\hat{\theta}_n$ is called a \textbf{consistent estimator} of $\theta$

    \vspace{0.8em}
    That is, for any $\varepsilon>0$: $\qquad P(|\hat{\theta}_n-\theta|>\varepsilon)\to 0\quad (n\to\infty)$

    \vspace{0.8em}
    \textbf{Interpretation:}
    \begin{itemize}
        \item With a small sample size, the estimator may have a large error
        \item As the sample size increases, the estimator should become increasingly stable and closer to the true value
    \end{itemize}

    \vspace{0.8em}
    \textbf{A Common Sufficient Condition:}\quad $E(\hat{\theta}_n)\to \theta\quad\text{and}\quad\mathrm{Var}(\hat{\theta}_n)\to 0$
% 則通常可推出 consistency。
\end{frame}

\begin{frame}{Example: Bernoulli Distribution}
Suppose
\vspace{-1.2em}
$$X_1, \dots,X_n \overset{iid}{\sim}\mathrm{Bernoulli}(\pi)$$
where $P(X_i=1)=\pi, \quad P(X_i=0)=1-\pi$

\vspace{0.8em}
The sample proportion is $\hat p=\bar X=\frac1n\sum_{i=1}^n X_i$ and $E(\hat p)=\pi,$
\,$\hat{p}$ is an unbiased estimator

\vspace{0.8em}
Also
\vspace{-1.2em}
$$\mathrm{Var}(X_i)=\pi(1-\pi)$$

\vspace{1.2em}
Hence
\vspace{-1.4em}
$$\mathrm{Var}(\hat p)=\frac{\pi(1-\pi)}{n}\to 0$$
Therefore
$$
\hat{p}
\overset{P}{\longrightarrow}
\pi
$$
So $\hat{p}$ is consistent

\vspace{0.8em}
\textbf{Furthermore:}
$$
T=\sum_{i=1}^n X_i \sim \mathrm{Binomial}(n, \pi)
$$
Since $\hat{p} = T/n$ is an unbiased function of $T$, it is the UMVUE of $\pi$
\end{frame}

\begin{frame}{From Estimation to SPC: The Core of Phase I}
    In Phase I SPC, we first need to establish an in-control baseline

    \vspace{0.5em}
    For an $\bar{X}$-chart, if $\mu_0$ and $\sigma_0$ are known, the control limits are
    $$
    UCL=\mu_0+z_{1-\alpha/2}\frac{\sigma_0}{\sqrt{m}},
    \quad
    LCL=\mu_0-z_{1-\alpha/2}\frac{\sigma_0}{\sqrt{m}}
    $$
    However, in practice these quantities are usually unknown, so we replace them with estimators $\hat{\mu}_0$ and $\hat{\sigma}_0$.
    
    \vspace{0.5em}
    Thus, the actual control limits become
    $$
    \widehat{UCL}=\hat{\mu}_0+z_{1-\alpha/2}\frac{\hat{\sigma}_0}{\sqrt{m}},
    \quad
    \widehat{LCL}=\hat{\mu}_0-z_{1-\alpha/2}\frac{\hat{\sigma}_0}{\sqrt{m}}
    $$
    \textbf{Potential issues:}
    \begin{itemize}
        \item If $\hat{\mu}_0$ is biased, the center line shifts.
        \item If $\hat{\sigma}_0$ has large variance, the control limits become unstable.
        \item If the estimators are not consistent, we cannot establish a reliable baseline.
    \end{itemize}
\end{frame}

\begin{frame}{How Estimator Quality Affects Monitoring: Example of a p-Chart}
    Suppose each subgroup has sample size $n$, and the defect rate is $\pi$
    Then $$\hat p=\frac{X}{n}, \quad X\sim\mathrm{Binomial}(n, \pi)$$
    In Phase I, we usually use $\bar{p}$ to estimate $\pi_0$ 

    The control limits of the p-chart are often written as

    $$
    UCL=\bar p + z\sqrt{\frac{\bar p(1-\bar p)}{n}},
    \quad
    LCL=\bar p - z\sqrt{\frac{\bar p(1-\bar p)}{n}}
    $$

    % \vspace{0.8em}
    % \textbf{這裡的統計邏輯是: }
    % \begin{enumerate}
    %     \item 先用樣本估計 $\pi_0$
    %     \item 再用估計的變異建立控制界限
    %     \item 最後判定新資料是否 out-of-control
    % \end{enumerate}
\end{frame}

\begin{frame}{Conclusion}
    \vspace{0.2cm}
    \textbf{(1) A good estimator should not be judged only by unbiasedness}
    \begin{itemize}
        \item $\mathrm{MSE}=\mathrm{Var}+\mathrm{Bias}^2$
    \end{itemize}
    % estimator 的品質來自偏差與波動的共同作用。

    \vspace{0.8em}
    \textbf{(2) Consistency is a requirement for long-run reliability}
    \begin{itemize}
        \item $\hat{\theta}_n \xrightarrow{P} \theta$
    \end{itemize}
    % 表示樣本變大後,估計量應穩定逼近真值。

    \vspace{0.8em}
    \textbf{(3) In SPC, estimation is the foundation of monitoring}
    \begin{itemize}
        \item Phase I requires estimation of $\mu_0$ and $\sigma_0$ in order to determine control limits
        \item Control limits determine false alarm rates and detection power
    \end{itemize}
    \vspace{0.8em}
    \uncover<2->{$$\boxed{\begin{array}{c}\text{If the baseline estimation is unreliable,} \\\text{then the entire SPC monitoring system becomes unreliable.}\end{array}}$$}
\end{frame}
