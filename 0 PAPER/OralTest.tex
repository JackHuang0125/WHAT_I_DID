\begin{frame}{\textbf{局部刻劃方法}}
    \vspace{0.5cm}
  % 不直接在原樣本附近做隨機擾動，而是以
    \begin{itemize}
    \item 在群 c 內，取滿足 $|\hat{p}-0.5|\le 0.05$ 的樣本集合，記為 $S^c_n$
    \item 對 $S^c_n$ 內的樣本特徵向量取平均，定義群內中心點為 $$\mathbf{x}^c_\mathbf{center}=\frac{1}{|S^c_n|}\sum_{i\in S^c_n}\mathbf{x}_i=(x^c_1,\dots,x^c_p)$$ 
    \item 以中心點為基準生成格子點： $$\mathbf{x}^c_{\mathbf{m}}=\mathbf{x}^c_\mathbf{center}+\delta \mathbf{m}$$ 其中 
      \begin{itemize}
        \item $\delta$ 為各維度相同的位移步長
        \item $\mathbf{m}=(m_1,\dots,m_p)$ 為位移向量，且$m_j\in\{-k,-k+1,\dots,0,\dots,k-1,k\}$
      \end{itemize}
    \item 共有 $(2k+1)^p$ 個格子點
    \end{itemize}
\end{frame}

\begin{frame}{\textbf{局部格子點舉例}}
  \centering
		\begin{tikzpicture}[scale=8]
			% vertical dashed lines
			\foreach \x in {1.0,1.1,1.2,1.3,1.4}{

  \huge{結論}

\begin{frame}{\textbf{結論}}
  \begin{itemize}
    \item 建立一套從資料前處理、分群、建模到局部線性刻劃的分析流程
    \vspace{1cm}
    \item 先分群再建模，可避免不同型態樣本的判斷規則混合，提升後續局部刻劃的清晰性
    \vspace{1cm}
    \item 在設定的局部範圍內，各群預測模型可藉由線性模型作一定程度的刻劃
    % \item 本研究提出一套適用於高維、缺失嚴重且類別不平衡製程資料的分析流程
    % \item 透過先分群再於群內建模，可降低異質性對模型學習與局部解釋的干擾
    % \item 以決策邊界附近樣本建立中心點，可聚焦於模型預測最不明確的局部範圍
    % \item 透過中心點與格子點設計，在可控制的局部範圍內擬合模型，可在較低外插風險下建立局部線性刻劃模型
    % \item 不同子群對應不同的預測模型之局部決策結構，分群的局部刻劃模型具分析價值
  \end{itemize}
\end{frame}
