  \begin{enumerate}[(1)]
    \item 目標變數重新編碼：Pass（$-$1 $\to$ 0），移除時間欄位
    \item 移除缺失值數量超過 1000 筆的單一特徵
    \item 移除相異值種類低於 10 種的單一特徵
    \item 採用 KNN imputer 進行補值，設定 $k=10$
    \item 移除離群樣本
  \end{enumerate}

\begin{frame}{\textbf{局部刻劃方法}}
    \vspace{0.5cm}
  % 不直接在原樣本附近做隨機擾動，而是以
    \begin{itemize}
    \item 在群 c 內，取滿足 $|\hat{p}-0.5|\le 0.05$ 的樣本集合，記為 $S^c_n$
    \item 對 $S^c_n$ 內的樣本特徵向量取平均，定義群內中心點為 $$\mathbf{x}^c_\mathbf{center}=\frac{1}{|S^c_n|}\sum_{i\in S^c_n}\mathbf{x}_i=(x^c_1,\dots,x^c_p)$$ 
    \item 以中心點為基準生成格子點： $$\mathbf{x}^c_{\mathbf{m}}=\mathbf{x}^c_\mathbf{center}+\delta \mathbf{m}$$ 其中 
      \begin{itemize}
        \item $\delta$ 為各維度相同的位移步長
        \item $\mathbf{m}=(m_1,\dots,m_p)$ 為位移向量，且$m_j\in\{-k,-k+1,\dots,0,\dots,k-1,k\}$
      \end{itemize}
    \item 共有 $(2k+1)^p$ 個格子點
    \end{itemize}
\end{frame}
