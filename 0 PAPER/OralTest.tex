\section{問題與目的}

\begin{frame}{\textbf{研究目的}}
  \begin{itemize}
    \item 若整體資料存在多個子群，單一模型的局部解釋可能混合不同規則。
      \item 分群後建立黑箱模型，以降低資料異質性對模型學習的干擾。
    \item 若以隨機擾動生成鄰近點，可能偏離真實資料分布，產生外插風險。
      \item 設計格子點生成方法，以控制局部擾動範圍，提升局部擬合的穩定性，並降低外插可能造成的偏誤
      \item 建立可檢視的局部刻劃模型，以作為黑箱模型在分類行為上的可解釋依據。
  \end{itemize}
\end{frame}

\section{方法與流程}

\begin{frame}{\textbf{流程大綱}}
\begin{enumerate}[(1)]
  \item 完成資料前處理，取得特徵矩陣 $X$ 與標籤 $y$。
  \item 以 SMOTE 後的資料執行 PCA，並在 PCA 空間做分群。
  \item 移除合成樣本後，於各群內訓練 XGBoost 黑箱分類模型。
  \item 挑選接近決策邊界的樣本，形成各群中心點。
  \item 在中心點附近生成局部格子點，並以黑箱輸出機率作為應變數。
  \item 建立群內局部線性模型，刻劃黑箱在局部區域的反應。
\end{enumerate}
% 建議此頁置入論文中的流程圖或自行重畫簡化流程圖
\end{frame}

\begin{frame}{\textbf{分群}}
\textbf{分群設計}
\begin{itemize}
  \item 以 SMOTE 補強少數類別於降維空間中的資訊。
  \item 於 PCA 空間以 k-means 進行分群。
  \item 以 elbow method 選定群數 $k=2$。
  \item 完成分群後移除合成樣本，只保留真實資料供後續建模。
\end{itemize}
\end{frame}

\begin{frame}{\textbf{黑箱模型}}
\textbf{黑箱模型}
\begin{itemize}
  \item 模型：XGBClassifier。
  \item 評估方式：五折交叉驗證。
  \item 僅在訓練集執行 SMOTE，測試集維持原分布。
  \item 評估指標：ACC、FAR、ROC-AUC。
\end{itemize}
\end{frame}

\begin{frame}{\textbf{局部刻劃方法}}
\begin{enumerate}[(1)]
  \item 在各群內以 $|\hat p_{test}-0.5|\le 0.05$ 挑選樣本點。
  \item 將候選樣本的標準化特徵向量取平均，形成群內中心點 $x_0$。
  \item 在 $x_0$ 附近依固定步長 $\delta$ 建立結構化格子點，而非隨機擾動樣本。
  \item 以黑箱模型對格子點輸出的預測機率作為應變數，建立 OLS 局部線性模型。
  \item 重複 50 次局部抽樣與篩選，統計穩定保留特徵；最後再以 200 萬筆格子點重新擬合局部模型。
\end{enumerate}
\end{frame}

\begin{frame}{\textbf{局部格子點設計概念}}
  \begin{block}{核心想法}
  不直接在原樣本附近做隨機擾動，而是以群內代表中心點為核心，建立可控制、可檢視的局部鄰域。
  \end{block}
  \begin{itemize}
    \item 每一維設定固定步長 $\delta$，位移倍數為 $m_j\in\{-k,-k+1,\dots,0,\dots,k\}$。
    \item 格子點表示為
    \[
      x^{(m)} = x_0 + \delta m.
    \]
    \item 本研究設定 $p=459$、$k=2$，理論格子點數為 $(2k+1)^p = 5^{459}$，無法完整列舉。
    \item 因此實作上改採隨機抽樣格子點建立局部模型。
  \end{itemize}
\end{frame}

\begin{frame}{\textbf{實驗環境與評估}}
\begin{itemize}
  \item 黑箱模型參數：\texttt{n\_estimators=300}、\texttt{max\_depth=3}、\texttt{learning\_rate=0.1}。
  \item 交叉驗證完成後，對每筆測試樣本保留黑箱輸出機率。
  \item 為補強五折平均值的穩定性說明，另以 bootstrap 方式建立 ACC、FAR、ROC-AUC 的 95\% 信賴區間。
  \item 局部刻劃模型以 $R^2$ 作為局部線性近似黑箱輸出的擬合度指標。
\end{itemize}
\end{frame}

\section{結果與討論}

\begin{frame}{\textbf{分群後資料概況}}
\begin{center}
\begin{tabular}{lcc}
\toprule
類別 & 群 0 & 群 1 \\
\midrule
Pass & 619 & 811 \\
Fail & 57 & 38 \\
總數 & 676 & 849 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.4em}
\begin{itemize}
  \item 兩群皆同時包含 Pass 與 Fail 樣本，可分別建立群內二元分類模型。
  \item 分群後並未使少數類別完全消失，後續局部刻劃仍可在各群內進行。
\end{itemize}
% 建議此頁可搭配論文圖 3.3
\end{frame}

\begin{frame}{\textbf{黑箱模型分類結果}}
\begin{center}
\small
\begin{tabular}{lccc}
\toprule
群別 & ACC & FAR & ROC-AUC \\
\midrule
全體 & 0.9272 & 0.0189 & 0.6636 \\
群 0 & 0.8847 & 0.0468 & 0.6974 \\
群 1 & 0.9482 & 0.0099 & 0.6266 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3em}
\begin{itemize}
  \item 群 0 與群 1 在分類表現上具有差異，尤其 FAR 差異明顯。
  \item 兩群 FAR 約相差 $0.0468/0.0099\approx 4.73$ 倍，顯示資料存在異質性。
  \item 若只建立整體單一模型，可能稀釋群 0 較高錯誤回報率的現象。
\end{itemize}
\end{frame}

\begin{frame}{\textbf{決策邊界樣本與中心點建立}}
\begin{itemize}
  \item 本研究以分類門檻 0.5 作為決策邊界基準。
  \item 選取條件為
  \[
    |\hat p_{test}-0.5|\le 0.05.
  \]
  \item 篩選結果：群 0 共 12 個候選樣本；群 1 共 7 個候選樣本。
  \item 將各群候選樣本取平均後形成中心點，再以中心點建立局部格子點鄰域。
\end{itemize}
\end{frame}

\begin{frame}{\textbf{局部刻劃模型結果}}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{群 0}
\begin{itemize}
  \item 50 次重複篩選後，穩定保留 96 個特徵。
  \item 以 200 萬筆局部格子點重新擬合。
  \item 局部線性模型 $R^2 = 0.58$。
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{群 1}
\begin{itemize}
  \item 50 次重複篩選後，穩定保留 56 個特徵。
  \item 同樣以 200 萬筆局部格子點重新擬合。
  \item 論文結果顯示群 1 的局部刻劃能力略優於群 0。
\end{itemize}
\end{column}
\end{columns}
\vspace{0.4em}
\begin{block}{解讀}
兩群的重要特徵組成與係數方向並不一致，代表不同子群的黑箱局部決策結構不同。
\end{block}
\end{frame}

\begin{frame}{\textbf{結果總結}}
\begin{itemize}
  \item 分群後，各群黑箱模型呈現不同的錯誤型態與可分類程度。
  \item 以決策邊界附近樣本建立中心點，可聚焦於模型判斷最不明確的局部區域。
  \item 結構化格子點可在可控制的鄰域內近似黑箱輸出，避免隨機擾動造成的外插偏誤。
  \item 群內局部刻劃模型可將黑箱決策轉為具體的局部重要特徵與方向性資訊。
\end{itemize}
\end{frame}

\begin{frame}{\textbf{結論}}
\begin{itemize}
  \item 本研究提出一套適用於高維、缺失嚴重且類別不平衡製程資料的分析流程。
  \item 透過先分群再於群內建模，可降低異質性對黑箱學習與局部解釋的干擾。
  \item 透過中心點與格子點設計，可在較低外插風險下建立局部線性刻劃模型。
  \item 研究結果顯示，不同子群對應不同的黑箱局部決策結構，分群式局部刻劃具分析價值。
\end{itemize}
\end{frame}
