\section*{Q1: $A(\eta)=-n\ln p_0=n\ln[1+\displaystyle\sum_{i=1}^s(e^{\eta_i})]$}
Since  $X_i {\sim}Multinomial(n;\ p_1,\ p_2,\ ...,\ p_s)$, it follows that   
$$p(X=x)=\frac{n!}{\displaystyle\prod_{i=0}^sx_i!}\prod_{i=0}^sp_i^{x_i}$$ 
and 
$\displaystyle\sum_{i=0}^sx_i=n$, 
$\displaystyle\sum_{i=0}^sp_i=1$. 
Moreover, we define $\eta_i=\ln\frac{p_i}{p_0}$, then the pmf can be written as 

\begin{align*}
p(X=x)
=\frac{n!}{\displaystyle\prod_{i=0}^sx_i!}\prod_{i=0}^sp_i^{x_i}
=\frac{n!}{\displaystyle\prod_{i=0}^sx_i!}p_0^{x_0}\prod_{i=1}^sp_i^{x_i}
=\frac{n!}{\displaystyle\prod_{i=0}^sx_i!}p_0^{n-\sum_{i=1}^sx_i}\prod_{i=1}^s(p_0e^{\eta_i})^{x_i},
\end{align*}
where
\[
\begin{aligned}
p_0^{n-\sum_{i=1}^sx_i}
&= \exp\left\{(n-\sum_{i=1}^sx_i)\ln p_0\right\}=\exp\left\{n\ln{p_0}-(\sum_{i=1}^sx_i)\ln{p_0}\right\}, \\
\prod_{i=1}^s (p_0 e^{\eta_i})^{x_i}
&= \exp\left\{\sum_{i=1}^s\eta_ix_i+(\sum_{i=1}^sx_i)\ln{p_0}\right\}.
\end{aligned}
\]
Then we have
$$
p(X=x)=\underbrace{\frac{n!}{\displaystyle\prod_{i=0}^sx_i!}}_{h(x)}
\exp\left\{\sum_{i=1}^s\underbrace{\eta_i x_i}_{\eta_i T_i(x)}
+\underbrace{n\ln p_0}_{-A(\eta)}\right\}
$$
Therefor, $A(\eta)=-n\ln{p_0}=-n\ln \frac{1}{1+\sum_{i=1}^s e^{\eta_i}}=n\ln (1+\displaystyle\sum_{i=1}^se^{\eta_i})$
where $\eta_i=\ln\frac{p_i}{p_0}
\implies p_i=p_0e^{\eta_i}
\implies \displaystyle p_0+\sum_{i=1}^sp_0e^{\eta_i}=1
\implies p_0=(1+\sum_{i=1}^se^{\eta_i})^{-1}$
